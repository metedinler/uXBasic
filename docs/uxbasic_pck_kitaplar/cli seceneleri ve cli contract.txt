Private Function UXBPathEndsWithAdim41(ByRef p As String, ByRef ext As String) As Integer
    If ext = "" Then Return 1
    Dim lp As String = LCase(Trim(p))
    Dim le As String = LCase(Trim(ext))
    If Left(le, 1) <> "." Then le = "." & le
    If Len(lp) < Len(le) Then Return 0
    Return Right(lp, Len(le)) = le
End Function

Private Function UXBCliKindFromOptionAdim41(ByRef optName As String) As String
    Dim u As String = LCase(Trim(optName))
    If InStr(u, "token") > 0 Or InStr(u, "lexer") > 0 Or InStr(u, "preprocessed") > 0 Then Return "LEXER"
    If InStr(u, "parser") > 0 Or InStr(u, "parse") > 0 Then Return "PARSER"
    If InStr(u, "ast") > 0 Then Return "AST"
    If InStr(u, "semantic") > 0 Or InStr(u, "symbol") > 0 Or InStr(u, "type-table") > 0 Or InStr(u, "layout") > 0 Or InStr(u, "hir") > 0 Then Return "SEMANTIC"
    If InStr(u, "mir") > 0 Then Return "MIR"
    If InStr(u, "x64") > 0 Or InStr(u, "x86") > 0 Or InStr(u, "nasm") > 0 Or InStr(u, "build") > 0 Then Return "NATIVE"
    If InStr(u, "js") > 0 Then Return "JS"
    If InStr(u, "wat") > 0 Then Return "WAT"
    If InStr(u, "wasm") > 0 Then Return "WASM"
    If InStr(u, "html") > 0 Or InStr(u, "manifest") > 0 Or InStr(u, "browser") > 0 Then Return "BROWSER"
    If InStr(u, "stdout") > 0 Or InStr(u, "stderr") > 0 Or InStr(u, "runtime") > 0 Or InStr(u, "run-report") > 0 Then Return "RUNTIME"
    If InStr(u, "source-map") > 0 Or InStr(u, "doc") > 0 Or InStr(u, "release") > 0 Or InStr(u, "validate") > 0 Then Return "TOOLING"
    Return "GENERAL"
End Function

Private Function UXBCliRequiredExtensionAdim41(ByRef optName As String) As String
    Dim u As String = LCase(Trim(optName))
    If InStr(u, "json") > 0 Then Return ".json"
    If InStr(u, "js-out") > 0 Or u = "--js-out" Then Return ".js"
    If InStr(u, "wat-out") > 0 Or u = "--wat-out" Or u = "--wasm-wat-out" Then Return ".wat"
    If u = "--wasm-out" Then Return ".wasm"
    If u = "--html-out" Then Return ".html"
    If u = "--manifest-out" Then Return ".json"
    If InStr(u, "nasm-out") > 0 Then Return ".asm"
    If u = "--build-x64-out" Then Return ".exe"
    If InStr(u, "stdout") > 0 Or InStr(u, "stderr") > 0 Or InStr(u, "preprocessed") > 0 Or InStr(u, "program-output-out") > 0 Then Return ".txt"
    If InStr(u, "log-out") > 0 Then Return ".log"

# Python gate script
gate_py = r'''#!/usr/bin/env python3
# ADIM41 CLI full contract gate
from __future__ import annotations
import argparse, csv, json, os, re, subprocess, sys
from pathlib import Path

PLACEHOLDER_PATTERNS = [
    "GENERATED_PLACEHOLDER",
    "artifact path reserved",
    "uxb-placeholder-artifact",
]

def read_matrix(path: Path):
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))

def expected_ext(row):
    opt = row["long"].lower()
    if "json" in opt:
        return ".json"
    if opt in ("--js-out",):
        return ".js"
    if opt in ("--wat-out", "--wasm-wat-out"):
        return ".wat"
    if opt == "--wasm-out":
        return ".wasm"
    if opt == "--html-out":
        return ".html"
    if opt == "--manifest-out":
        return ".json"
    if opt == "--emit-x64-nasm-out":
        return ".asm"
    if opt == "--build-x64-out":
        return ".exe"
    if opt in ("--stdout-out", "--stderr-out", "--preprocessed-out", "--program-output-out"):
        return ".txt"
    if "log-out" in opt:
        return ".log"
    return ""

def classify(row):
    opt = row["long"].lower()
    if opt in ("--ayikla","--sessiz","--semantik"):
        return "REMOVED_TURKISH_ALIAS"
    if opt.endswith("-out") or "json-out" in opt or opt == "--trace-json":
        return "FILE_OUTPUT"
    return "FLAG_OR_MODE"

def validate_generated_file(path: Path, ext: str):
    if not path.exists():
        return False, "missing"
    if path.stat().st_size == 0:
        return False, "empty"
    txt = ""
    if path.suffix.lower() in (".json",".txt",".log",".js",".wat",".html",".css",".md",".asm"):
        txt = path.read_text(encoding="utf-8", errors="ignore")
        for p in PLACEHOLDER_PATTERNS:
            if p in txt:
                return False, "placeholder"
    if ext == ".json":
        try:
            json.loads(txt)
        except Exception as e:
            return False, f"bad_json:{e}"
    return True, "ok"

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--matrix", required=True)
    ap.add_argument("--uxb-exe", required=False, default="")
    ap.add_argument("--source", required=False, default="")
    ap.add_argument("--outdir", required=True)
    ap.add_argument("--report-json", required=True)
    ap.add_argument("--report-csv", required=True)
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()

    matrix = read_matrix(Path(args.matrix))
    outdir = Path(args.outdir)
    outdir.mkdir(parents=True, exist_ok=True)

    results = []
    short_seen = {}
    long_seen = {}
    ok = True

    for row in matrix:
        long = row["long"]
        short = row["short"]
        kind = classify(row)
        ext = expected_ext(row)
        item_ok = True
        notes = []

        if short in short_seen:
            item_ok = False
            notes.append(f"duplicate_short:{short_seen[short]}")
        short_seen[short] = long

        if long in long_seen:
            item_ok = False
            notes.append("duplicate_long")
        long_seen[long] = True

        if kind == "REMOVED_TURKISH_ALIAS":
            notes.append("must_fail_if_used")
        elif kind == "FILE_OUTPUT":
            if ext == "":
                item_ok = False
                notes.append("file_output_without_expected_ext")
            # This gate can run in dry mode when compiler not ready.
            # In active mode it verifies existing output if present.
            sample = outdir / (long.strip("-").replace("-","_") + (ext or ".out"))
            if sample.exists():
                f_ok, f_note = validate_generated_file(sample, ext)
                item_ok = item_ok and f_ok
                notes.append(f_note)
            else:
                notes.append("not_executed_sample_missing")
        else:
            notes.append("flag_registry_only")

        if not item_ok:
            ok = False
        results.append({
            "no": row["no"], "long": long, "short": short, "kind": kind,
            "expected_ext": ext, "ok": item_ok, "notes": ";".join(notes),
            "vscode": row.get("vscode","")
        })

    with Path(args.report_csv).open("w", encoding="utf-8", newline="") as f:
        w = csv.DictWriter(f, fieldnames=list(results[0].keys()))
        w.writeheader()
        w.writerows(results)

    report = {
        "schema": "uxb-adim41-cli-full-contract-gate-1",
        "ok": ok,
        "command_count": len(matrix),
        "removed_turkish_alias_count": sum(1 for r in results if r["kind"] == "REMOVED_TURKISH_ALIAS"),
        "file_output_count": sum(1 for r in results if r["kind"] == "FILE_OUTPUT"),
        "duplicate_short_count": len([r for r in results if "duplicate_short" in r["notes"]]),
        "failed_count": len([r for r in results if not r["ok"]]),
        "dry_run": args.dry_run,
        "reports": [str(args.report_csv)],
    }
    Path(args.report_json).write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
    return 0 if ok else 1

if __name__ == "__main__":
    raise SystemExit(main())
'''
(tools / "uxb_cli_full_contract_gate.py").write_text(gate_py, encoding="utf-8")

function Find-UxbRoot {
    $p = (Get-Location).Path
    while ($true) {
        if ((Test-Path (Join-Path $p "src")) -and (Test-Path (Join-Path $p "compiler"))) { return $p }
        $parent = Split-Path $p -Parent
        if ($parent -eq $p) { throw "uXBasic root not found" }
        $p = $parent
    }
}

$root = Find-UxbRoot
Set-Location $root

$current = "reports\control\current"
New-Item -ItemType Directory -Force -Path $current | Out-Null

$matrix = "docs\ADIM41_CLI_COMMAND_MATRIX.csv"
if (!(Test-Path $matrix)) {
    if (Test-Path "UXB_KATMAN_BAZLI_COPILOT_BELGE_SETI\docs\ADIM41_CLI_COMMAND_MATRIX.csv") {
        $matrix = "UXB_KATMAN_BAZLI_COPILOT_BELGE_SETI\docs\ADIM41_CLI_COMMAND_MATRIX.csv"
    } else {
        throw "ADIM41_CLI_COMMAND_MATRIX.csv not found"
    }
}

$py = "tools\control\uxb_cli_full_contract_gate.py"
if (!(Test-Path $py)) { throw "uxb_cli_full_contract_gate.py not found" }

$reportJson = Join-Path $current "adim41_cli_full_contract_gate.json"
$reportCsv  = Join-Path $current "adim41_cli_full_contract_results.csv"
$outdir     = Join-Path $current "adim41_cli_outputs"
New-Item -ItemType Directory -Force -Path $outdir | Out-Null

python $py `
  --matrix $matrix `
  --outdir $outdir `
  --report-json $reportJson `
  --report-csv $reportCsv `
  --dry-run

$code = $LASTEXITCODE

Write-Output "ADIM41 CLI full contract gate exit code: $code"
Write-Output "Report: $reportJson"
Write-Output "CSV: $reportCsv"

exit $code
'''
(scripts / "run_adim41_cli_full_contract_gate.ps1").write_text(ps1, encoding="utf-8")
