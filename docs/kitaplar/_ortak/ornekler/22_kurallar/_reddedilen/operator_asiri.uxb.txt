' KONU: OPERATOR + (kullanıcı operatörü)
' KOMUTLAR: CLASS OPERATOR
' ACIKLAMA: Sınıf, OPERATOR bildirimiyle + gibi operatörleri yeniden tanımlar.
'= 4
CLASS Sayi
    PUBLIC
    DIM v AS I32
    OPERATOR +(other AS Sayi) AS Sayi
        DIM r AS Sayi = NEW Sayi()
        r.v = v + other.v
        RETURN r
    END OPERATOR
END CLASS

DIM a AS Sayi = NEW Sayi()
DIM b AS Sayi = NEW Sayi()
a.v = 1
b.v = 3
DIM c AS Sayi = a + b
PRINT c.v
